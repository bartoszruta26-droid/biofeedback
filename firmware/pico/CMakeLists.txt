cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED PICO_SDK_PATH)
    set(PICO_SDK_PATH $ENV{PICO_SDK_PATH})
endif()
if(NOT PICO_SDK_PATH)
    message(FATAL_ERROR "PICO_SDK_PATH must point to the Raspberry Pi Pico SDK")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(biofeedback_pico_firmware C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
pico_sdk_init()
add_executable(biofeedback_pico main.cpp)
target_link_libraries(biofeedback_pico pico_stdlib hardware_uart hardware_watchdog)
pico_enable_stdio_usb(biofeedback_pico 1)
pico_enable_stdio_uart(biofeedback_pico 0)
pico_add_extra_outputs(biofeedback_pico)
